sbash () { 
    srun --pty --x11 $@ /bin/bash
}

sibash () { 
    srun --pty --x11 --interactive $@ /bin/bash
}

alias sqme="squeue --me"

## grep on history
function hgrep {
    history | grep "$@"
}

function Rnosave {
  x="$1"
  tempfile=`mktemp file.XXXX.sh`
  echo "#!/bin/bash" > $tempfile
  # need to load bash_profile due to sbatch
  echo ". ~/.bash_profile" >> $tempfile
  echo "R --no-save < ${x}" >> $tempfile
  shift
  cmd="sbatch $@ $tempfile"
  echo "cmd is $cmd"
  ${cmd}
  rm $tempfile
}
