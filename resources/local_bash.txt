# Replace JHPCE_USERNAME with your JHPCE username

alias jhpce="ssh -Y -X JHPCE_USERNAME@jhpce01.jhsph.edu"
# connect to SLURM, run interactive session  
alias sqr="jhpce -t 'source /etc/profile; srun --pty --x11 $@ /bin/bash"
## grep on history
function hgrep {
    history | grep "$@"
}
# change dirs to directory of file
function cdd { 
    x=`dirname "${1}"`;
    cd "${x}"
} 

# connect straight to node with certain gig memory.  Example:
sqrgig 30
function sqrgig { 
    x="$1";
    x="${x}G";
    shift;
    ell="--mem=$x";
    echo "jhpce requests were: $ell $@";
    cmd="source ~/.bash_profile; srun--pty --x11 $@ $ell /bin/bash";
    jhpce -t $cmd
}
